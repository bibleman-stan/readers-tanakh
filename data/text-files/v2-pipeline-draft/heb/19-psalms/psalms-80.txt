80:1
לַמְנַצֵּ֥חַ
אֶל־שֹׁשַׁנִּ֑ים
עֵד֖וּת לְאָסָ֣ף
מִזְמֹֽור׃

80:2
רֹ֘עֵ֤ה יִשְׂרָאֵ֨ל׀
הַאֲזִ֗ינָה נֹהֵ֣ג כַּצֹּ֣אן יֹוסֵ֑ף יֹשֵׁ֖ב הַכְּרוּבִ֣ים
הֹופִֽיעָה׃

80:3
לִפְנֵ֤י אֶפְרַ֨יִם׀ וּבִנְיָ֘מִ֤ן וּמְנַשֶּׁ֗ה עֹורְרָ֥ה אֶת־גְּבֽוּרָתֶ֑ךָ
וּלְכָ֖ה לִישֻׁעָ֣תָה לָּֽנוּ׃

80:4
אֱלֹהִ֥ים
הֲשִׁיבֵ֑נוּ
וְהָאֵ֥ר פָּ֝נֶ֗יךָ
וְנִוָּשֵֽׁעָה׃

80:5
יְהוָ֣ה אֱלֹהִ֣ים צְבָאֹ֑ות
עַד־מָתַ֥י עָ֝שַׁ֗נְתָּ בִּתְפִלַּ֥ת עַמֶּֽךָ׃

80:6
הֶ֭אֱכַלְתָּם לֶ֣חֶם דִּמְעָ֑ה
וַ֝תַּשְׁקֵ֗מֹו בִּדְמָעֹ֥ות שָׁלִֽישׁ׃

80:7
תְּשִׂימֵ֣נוּ מָ֭דֹון לִשְׁכֵנֵ֑ינוּ
וְ֝אֹיְבֵ֗ינוּ יִלְעֲגוּ־לָֽמֹו׃

80:8
אֱלֹהִ֣ים צְבָאֹ֣ות
הֲשִׁיבֵ֑נוּ
וְהָאֵ֥ר פָּ֝נֶ֗יךָ
וְנִוָּשֵֽׁעָה׃

80:9
גֶּ֭פֶן מִמִּצְרַ֣יִם תַּסִּ֑יעַ תְּגָרֵ֥שׁ גֹּ֝ויִ֗ם
וַתִּטָּעֶֽהָ׃

80:10
פִּנִּ֥יתָ לְפָנֶ֑יהָ
וַתַּשְׁרֵ֥שׁ שָׁ֝רָשֶׁ֗יהָ
וַתְּמַלֵּא־אָֽרֶץ׃

80:11
כָּסּ֣וּ הָרִ֣ים צִלָּ֑הּ
וַ֝עֲנָפֶ֗יהָ אַֽרְזֵי־אֵֽל׃

80:12
תְּשַׁלַּ֣ח קְצִירֶ֣הָ עַד־יָ֑ם
וְאֶל־נָ֝הָ֗ר יֹֽונְקֹותֶֽיהָ׃

80:13
לָ֭מָּה פָּרַ֣צְתָּ גְדֵרֶ֑יהָ
וְ֝אָר֗וּהָ כָּל־עֹ֥בְרֵי דָֽרֶךְ׃

80:14
יְכַרְסְמֶ֣נָּֽה חֲזִ֣יר מִיָּ֑עַר
וְזִ֖יז שָׂדַ֣י יִרְעֶֽנָּה׃

80:15
אֱלֹהִ֣ים צְבָאֹות֮
שֽׁ֫וּב־נָ֥א
הַבֵּ֣ט מִשָּׁמַ֣יִם
וּרְאֵ֑ה
וּ֝פְקֹ֗ד גֶּ֣פֶן זֹֽאת׃

80:16
וְ֭כַנָּה אֲשֶׁר־נָטְעָ֣ה יְמִינֶ֑ךָ
וְעַל־בֵּ֝֗ן אִמַּ֥צְתָּה לָּֽךְ׃

80:17
שְׂרֻפָ֣ה בָאֵ֣שׁ
כְּסוּחָ֑ה
מִגַּעֲרַ֖ת פָּנֶ֣יךָ יֹאבֵֽדוּ׃

80:18
תְּֽהִי־יָ֭דְךָ עַל־אִ֣ישׁ יְמִינֶ֑ךָ
עַל־בֶּן־אָ֝דָ֗ם אִמַּ֥צְתָּ לָּֽךְ׃

80:19
וְלֹא־נָסֹ֥וג מִמֶּ֑ךָּ תְּ֝חַיֵּ֗נוּ
וּבְשִׁמְךָ֥ נִקְרָֽא׃

80:20
יְה֘וָ֤ה אֱלֹהִ֣ים צְבָאֹ֣ות
הֲשִׁיבֵ֑נוּ
הָאֵ֥ר פָּ֝נֶ֗יךָ
וְנִוָּשֵֽׁעָה׃
